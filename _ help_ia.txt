php/helpers/Functions.php;
<?php

function findById(array $array, string $campo, int $id): ?array
{
    foreach ($array as $item) {

        if ($item[$campo] == $id) {
            return $item;
        }

    }

    return null;
}

function findByForeignKey(array $array, string $campo, int $valor): array
{
    return array_values(
        array_filter(
            $array,
            fn($item) => $item[$campo] == $valor
        )
    );
}

function validateId(array $parts): int
{
    if (!isset($parts[1]) || !is_numeric($parts[1])) {
        Response::badRequest("ID inválido.");
    }

    return (int)$parts[1];
}



php/helpers/Response.php;

<?php

class Response
{
    public static function json($data, int $status = 200): void
    {
        http_response_code($status);

        echo json_encode(
            $data,
            JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE
        );

        exit;
    }

    public static function success($data): void
    {
        self::json($data, 200);
    }

    public static function badRequest(string $message): void
    {
        self::json([
            "erro" => $message
        ], 400);
    }

    public static function notFound(string $message): void
    {
        self::json([
            "erro" => $message
        ], 404);
    }

    public static function methodNotAllowed(): void
    {
        self::json([
            "erro" => "Método não permitido."
        ], 405);
    }
}

<?php

header("Access-Control-Allow-Origin: *");
header("Access-Control-Allow-Credentials: true");
header("Access-Control-Allow-Methods: GET, POST, PATCH, PUT, DELETE, OPTIONS");
header("Access-Control-Allow-Headers: Content-Type, Accept, Authorization, X-Requested-With, X-Auth-Token, Origin, Application");
header("Content-Type: application/json; charset=utf-8");

/*
|--------------------------------------------------------------------------
| CORS Preflight
|--------------------------------------------------------------------------
*/
if ($_SERVER['REQUEST_METHOD'] === 'OPTIONS') {
    http_response_code(204);
    exit;
}



/*
|--------------------------------------------------------------------------
| Helpers
|--------------------------------------------------------------------------
*/
require_once __DIR__ . '/helpers/Response.php';
require_once __DIR__ . '/helpers/Functions.php';

/*
|--------------------------------------------------------------------------
| Função utilitária (persistência)
|--------------------------------------------------------------------------
*/
function saveAccounts($file, $data) {
    file_put_contents(
        $file,
        json_encode($data, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE)
    );
}


/*
|--------------------------------------------------------------------------
| Dados
|--------------------------------------------------------------------------
*/
$banks = require __DIR__ . "/data/banks.php";
$accountsFile = __DIR__ . "/data/accounts.json";
$accounts = json_decode(file_get_contents($accountsFile), true) ?? [];

$reports = require __DIR__ . "/data/reports.php";
$cards   = require __DIR__ . "/data/creditcards.php";
$posts   = require __DIR__ . "/data/posts.php";

/*
|--------------------------------------------------------------------------
| URL parsing
|--------------------------------------------------------------------------
*/
$uri = parse_url($_SERVER["REQUEST_URI"], PHP_URL_PATH);

$parts = array_values(array_filter(explode("/", trim($uri, "/"))));

if (isset($parts[0]) && $parts[0] === "api") {
    array_shift($parts);
    $parts = array_values($parts);
}

if (empty($parts)) {
    Response::notFound("Endpoint não informado.");
}

/*
|--------------------------------------------------------------------------
| Router
|--------------------------------------------------------------------------
*/
switch ($parts[0]) {

    /*
    |--------------------------------------------------------------------------
    | BANKS
    |--------------------------------------------------------------------------
    */
    case "banks":

        // GET /banks
        if (count($parts) === 1) {
            Response::success($banks);
        }

        $bankId = validateId($parts);

        // GET /banks/{id}
        if (count($parts) === 2) {

            $bank = findById($banks, "id_bnk", $bankId);

            if (!$bank) {
                Response::notFound("Banco não encontrado.");
            }

            Response::success($bank);
        }

        // GET /banks/{id}/accounts
        if (isset($parts[2]) && $parts[2] === "accounts") {

            $result = findByForeignKey(
                $accounts,
                "fk_bank",
                $bankId
            );

            Response::success($result);
        }

        Response::notFound("Recurso não encontrado.");
        break;

    /*
    |--------------------------------------------------------------------------
    | ACCOUNTS
    |--------------------------------------------------------------------------
    */
    case "accounts":

        /*
        |--------------------------------------------------------------------------
        | POST /accounts (CRIAR CONTA)
        |--------------------------------------------------------------------------
        */
        if ($_SERVER["REQUEST_METHOD"] === "POST") {

            $input = json_decode(file_get_contents("php://input"), true);          

            $newAccount = [
                "id_act" => time(),
                "number_act" => $input["number_act"] ?? "",
                "type_act" => $input["type_act"] ?? "",
                "desc_act" => $input["desc_act"] ?? "",
                "saldo_act" => isset($input["saldo_act"]) ? (float)$input["saldo_act"] : 0,
                "fk_bank" => $input["fk_bank"] ?? null
            ];

            $accounts[] = $newAccount;

            saveAccounts($accountsFile, $accounts);

            Response::success($newAccount, 201);
        }

        $accountId = validateId($parts);

        // GET /accounts/{id}
        if (count($parts) === 2) {

            $account = findById($accounts, "id_act", $accountId);

            if (!$account) {
                Response::notFound("Conta não encontrada.");
            }

            Response::success($account);
        }

        // GET /accounts/{id}/reports
        if (isset($parts[2]) && $parts[2] === "reports") {

            Response::success(
                findByForeignKey($reports, "fk_act", $accountId)
            );
        }

        // GET /accounts/{id}/creditcards
        if (isset($parts[2]) && $parts[2] === "creditcards") {

            Response::success(
                findByForeignKey($cards, "fk_act", $accountId)
            );
        }

        Response::notFound("Recurso não encontrado.");
        break;

    /*
    |--------------------------------------------------------------------------
    | CREDIT CARDS
    |--------------------------------------------------------------------------
    */
    case "creditcards":

        $cardId = validateId($parts);

        // GET /creditcards/{id}
        if (count($parts) === 2) {

            $card = findById($cards, "id_ccr", $cardId);

            if (!$card) {
                Response::notFound("Cartão não encontrado.");
            }

            Response::success($card);
        }

        // GET /creditcards/{id}/posts
        if (isset($parts[2]) && $parts[2] === "posts") {

            Response::success(
                findByForeignKey($posts, "fk_ccr", $cardId)
            );
        }

        Response::notFound("Recurso não encontrado.");
        break;

    /*
    |--------------------------------------------------------------------------
    | DEFAULT
    |--------------------------------------------------------------------------
    */
    default:
        Response::notFound("Endpoint não encontrado.");
}
